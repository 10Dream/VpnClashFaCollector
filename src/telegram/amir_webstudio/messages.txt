cd cfst
wget -N https://github.com/XIU2/CloudflareSpeedTest/releases/download/v2.3.5/cfst_linux_arm64.tar.gz
tar -zxf cfst_linux_arm64.tar.gz
chmod +x cfst

`./cfst`

`./cfst -ip 199.181.197.0/24`

`./cfst -f list.txt`
